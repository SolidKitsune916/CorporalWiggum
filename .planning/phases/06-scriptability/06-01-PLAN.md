  <name>Task 1: Create exit codes and JSON output utilities</name>
  <files>cli/src/lib/exit-codes.ts, cli/src/lib/json-output.ts</files>
  <action>
Create two new utility files:

1. `exit-codes.ts` - Define standard exit codes:
```typescript
export const EXIT_CODES = {
  SUCCESS: 0,        // Command succeeded
  GENERAL_ERROR: 1,  // General error (catch-all)
  INVALID_USAGE: 2,  // Invalid arguments/options
  NOT_FOUND: 64,     // Resource not found (project, session)
  ALREADY_EXISTS: 65 // Resource conflict (loop already running)
} as const;

export type ExitCode = typeof EXIT_CODES[keyof typeof EXIT_CODES];
```

2. `json-output.ts` - JSON output helpers:
```typescript
export interface JsonOutput<T> {
  success: boolean;
  data?: T;
  error?: string;
  metadata: {
    timestamp: string;
    version: string;
  };
}

export function outputJson<T>(data: T): void {
  const output: JsonOutput<T> = {
    success: true,
    data,
    metadata: {
      timestamp: new Date().toISOString(),
      version: '3.0.0',
    },
  };
  console.log(JSON.stringify(output, null, 2));
}

export function outputJsonError(error: string): void {
  const output: JsonOutput<never> = {
    success: false,
    error,
    metadata: {
      timestamp: new Date().toISOString(),
      version: '3.0.0',
    },
  };
  console.log(JSON.stringify(output, null, 2));
}
```

Key: JSON errors go to stdout (not stderr) for consistent parsing. Include metadata for debugging.
