Modify InstanceSpawner to enrich the instance list with session data:

1. Import getSessionRepository at the top:
   ```typescript
   import { getSessionRepository } from './database/repositories/SessionRepository.js';
   ```

2. Modify listInstances() method to query session data for each running instance:
   ```typescript
   listInstances(): LauncherInstance[] {
     const sessionRepo = getSessionRepository();

     return Array.from(this.instances.values()).map(instance => {
       // Get active session for this project to include metrics
       const session = sessionRepo.getActiveSessionForProject(instance.projectId);

       return {
         projectId: instance.projectId,
         backendPort: instance.backendPort,
         frontendPort: instance.frontendPort,
         pid: instance.pid,
         startedAt: instance.startedAt,
         loopStatus: instance.loopStatus ? {
           ...instance.loopStatus,
           // Enrich with session data if available
           costSpent: session?.costSpent,
           maxIterations: session?.maxIterations ?? undefined,
           state: session?.state,
         } : undefined,
       };
     });
   }
   ```

3. Similarly update getInstance() method to include the same enrichment:
   ```typescript
   getInstance(projectId: string): LauncherInstance | null {
     const instance = this.instances.get(projectId);
     if (!instance) return null;

     const sessionRepo = getSessionRepository();
     const session = sessionRepo.getActiveSessionForProject(projectId);

     return {
       projectId: instance.projectId,
       backendPort: instance.backendPort,
       frontendPort: instance.frontendPort,
       pid: instance.pid,
       startedAt: instance.startedAt,
       loopStatus: instance.loopStatus ? {
         ...instance.loopStatus,
         costSpent: session?.costSpent,
         maxIterations: session?.maxIterations ?? undefined,
         state: session?.state,
       } : undefined,
     };
   }
   ```

Note: The session query is fast (single indexed lookup by projectId) and happens on each list request. This is acceptable for launcher use (infrequent calls, small instance counts).
