1. Define interfaces:

```typescript
export interface OrphanedLoop {
  projectId: string;
  projectPath?: string;
  pid: number;
  mode: LoopMode;
  startedAt: string;
  source: 'database' | 'pidfile';
  status: 'alive' | 'dead';
}

export interface OrphanDetectionResult {
  orphans: OrphanedLoop[];           // Live processes without proper tracking
  staleSessions: string[];           // Session IDs marked as crashed (dead processes)
  stalePidFiles: string[];           // PID files deleted (dead processes)
}
```

2. Implement OrphanDetector class:

   - Constructor: Takes PidFileManager, SessionRepository

   - `async detectOrphans(): Promise<OrphanDetectionResult>`
     Main detection logic:

     a. Get active sessions from database
     b. Get PID files from filesystem
     c. For each active session:
        - If process is dead: mark session crashed, add to staleSessions
        - If process is alive but no PID file: add to orphans (source: 'database')
     d. For each PID file:
        - If process is dead: delete PID file, add to stalePidFiles
        - If process is alive but no matching session: add to orphans (source: 'pidfile', status: 'alive')
     e. Return OrphanDetectionResult

   - `async cleanupOrphan(orphan: OrphanedLoop): Promise<boolean>`
     Kill an orphaned process:
     - Use GracefulShutdown.stopAndVerify()
     - Delete PID file if exists
     - Mark session crashed if exists
     - Return success/failure

   - `async cleanupAllOrphans(orphans: OrphanedLoop[]): Promise<{ cleaned: number; failed: number }>`
     Clean up multiple orphans

   - Private `isProcessAlive(pid: number)`: Signal 0 check (reuse pattern)

3. Import dependencies:
   - PidFileManager from './PidFileManager.js'
   - GracefulShutdown from './GracefulShutdown.js'
   - getSessionRepository from '../database/repositories/SessionRepository.js'
   - logger from '../lib/logger.js'
   - LoopMode type

Pattern for detection (from research):
```typescript
async detectOrphans(): Promise<OrphanDetectionResult> {
  const result: OrphanDetectionResult = {
    orphans: [],
    staleSessions: [],
    stalePidFiles: []
  };

  const sessions = this.sessionRepo.getActiveSessions();
  const pidFiles = await this.pidFileManager.listPidFiles();
  const pidFileMap = new Map(pidFiles.map(pf => [pf.pid, pf]));
  const sessionPids = new Set(sessions.map(s => s.pid));

  // Check sessions
  for (const session of sessions) {
    const alive = this.isProcessAlive(session.pid);
    if (!alive) {
      this.sessionRepo.markSessionCrashed(session.id);
      result.staleSessions.push(session.id);
    } else if (!pidFileMap.has(session.pid)) {
      result.orphans.push({
        projectId: session.projectId,
        pid: session.pid,
        mode: session.mode,
        startedAt: session.startedAt,
        source: 'database',
        status: 'alive'
      });
    }
  }

  // Check PID files
  for (const pidFile of pidFiles) {
    const alive = this.isProcessAlive(pidFile.pid);
    if (!alive) {
      await this.pidFileManager.deletePidFile(pidFile.projectId);
      result.stalePidFiles.push(pidFile.projectId);
    } else if (!sessionPids.has(pidFile.pid)) {
      result.orphans.push({
        projectId: pidFile.projectId,
        projectPath: pidFile.projectPath,
        pid: pidFile.pid,
        mode: pidFile.mode,
        startedAt: pidFile.startedAt,
        source: 'pidfile',
        status: 'alive'
      });
    }
  }

  return result;
}
```

```typescript
}
```

5. Add WebSocket message handlers for orphan cleanup:
```typescript
case 'orphans:cleanup':
  try {
    const { pid, projectId } = message.payload as { pid: number; projectId: string };
    const orphan = pendingOrphans.find(o => o.pid === pid && o.projectId === projectId);
    if (!orphan) {
      ws.send(JSON.stringify({
        type: 'orphans:error',
        payload: { error: 'Orphan not found' }
      }));
      break;
    }
    const success = await orphanDetector.cleanupOrphan(orphan);
    if (success) {
      pendingOrphans = pendingOrphans.filter(o => !(o.pid === pid && o.projectId === projectId));
      broadcast({ type: 'orphans:cleaned', payload: { pid, projectId } });
      broadcast({ type: 'orphans:detected', payload: { orphans: pendingOrphans } });
    } else {
      ws.send(JSON.stringify({
        type: 'orphans:error',
        payload: { error: 'Failed to cleanup orphan' }
      }));
    }
  } catch (err) {
    ws.send(JSON.stringify({
      type: 'orphans:error',
      payload: { error: err instanceof Error ? err.message : 'Unknown error' }
    }));
  }
  break;

case 'orphans:cleanup-all':
  try {
    const result = await orphanDetector.cleanupAllOrphans(pendingOrphans);
    pendingOrphans = [];
    broadcast({
      type: 'orphans:all-cleaned',
      payload: { cleaned: result.cleaned, failed: result.failed }
    });
    broadcast({ type: 'orphans:detected', payload: { orphans: [] } });
  } catch (err) {
    ws.send(JSON.stringify({
      type: 'orphans:error',
      payload: { error: err instanceof Error ? err.message : 'Unknown error' }
    }));
  }
  break;

case 'orphans:ignore':
  // User chose to ignore orphans (keep them running)
  pendingOrphans = [];
  broadcast({ type: 'orphans:detected', payload: { orphans: [] } });
  break;
```
